import assert from 'node:assert/strict';
import {readFile,writeFile} from 'node:fs/promises';
import {loadAuditLab} from '../../scripts/load-audit-lab.ts';
import {loadComparisons} from '../../scripts/load-comparisons.ts';
import {auditCandidate,chooseAcceptable} from '../../src/audit/evaluate.ts';
import {judgeMutation} from '../../src/audit/mutate.ts';
import {validatePreference} from '../../src/audit/preference.ts';
const state=await loadAuditLab(),run=JSON.parse(await readFile('docs/reviews/h6/run.json','utf8'));
const probes:string[]=[];
const mobile=run.receipts.find((r:any)=>r.label==='repair-before-mobile');
assert.equal(mobile.view.requested_viewport.width,390);
assert.equal(mobile.view.geometry.viewport.width,1308);
assert.equal(mobile.audit.viewport.width,390);
assert.ok(mobile.audit.issues.some((i:any)=>i.code==='OVERFLOW'&&i.severity==='BLOCKING'));
probes.push('Requested 390 vs observed 1308 remains explicit and blocking');
const baseline=run.receipts.filter((r:any)=>r.kind==='FINALIST');
const entries=state.mounts.filter(m=>m.kind==='FINALIST').map(m=>({candidate:m.candidate,reports:baseline.filter((r:any)=>r.mount_id===m.id).map((r:any)=>r.audit)}));
assert.equal(chooseAcceptable(entries).finalists.length,4);
const corrupted=structuredClone(entries);corrupted.forEach(e=>e.reports[0].candidate_sha256='0'.repeat(64));
assert.equal(chooseAcceptable(corrupted).status,'NO_ACCEPTABLE_DESIGN');
probes.push('Complete coverage with stale candidate hashes cannot pass selection');
const h=run.history[0],after=structuredClone(h.after_reports);after[0].policy_sha256='f'.repeat(64);
assert.throws(()=>judgeMutation(h.before,h.after,h.before_reports,after,h.evidence),/POLICY/);
assert.deepEqual(run.history[1].kept,run.history[1].before);
probes.push('Mutation policy mismatch rejects and rollback restores full prior candidate');
const view=structuredClone(baseline[0].view);view.observation.media[0].credit='';
const gate=auditCandidate(baseline[0].candidate,state.lab,view,state.policy);
assert.equal(gate.status,'FAIL');assert.ok(gate.issues.some(i=>i.family==='rights'&&i.severity==='BLOCKING'));
probes.push('Credit deletion blocks even with unchanged favorable raster');
const pair=(await loadComparisons()).pairs[0];
const record={schema_version:'1.0.0',source:'HUMAN_LOCAL_PANEL',comparison_id:pair.id,context_sha256:pair.context_sha256,left_id:pair.left_id,right_id:pair.right_id,left_sha256:pair.left_sha256,right_sha256:pair.right_sha256,left_screenshot_sha256:pair.left_screenshot_sha256,right_screenshot_sha256:pair.right_screenshot_sha256,semantic_page:pair.semantic_page,viewport:pair.viewport,decision:'NONE',reason:'REVIEW_SCHEMA_TEST_ONLY',recorded_at:'2026-10-09T00:00:00Z',acceptance_scope:'THIS_PAIR_ROUTE_VIEWPORT_ONLY'};
assert.ok(validatePreference(record,pair).success);
assert.ok(!validatePreference({...record,left_screenshot_sha256:'0'.repeat(64)},pair).success);
assert.ok(!validatePreference({...record,context_sha256:'0'.repeat(64)},pair).success);
assert.ok(!validatePreference({...record,source:'TEST_SIMULATION'},pair).success);
probes.push('Schema-only fixture tests screenshot/context drift and simulation rejection; not a human vote');
await writeFile('docs/reviews/h6/probes.json',JSON.stringify({status:'PASSED',source:'REVIEW_TEST_ONLY',human_votes:'NOT_COLLECTED',probes},null,2)+'\n');
console.log(JSON.stringify({status:'PASSED',independent_probes:probes.length,human_votes:'NOT_COLLECTED'}));
